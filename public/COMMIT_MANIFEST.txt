+  const env = project.environment;

+    applies = false;

+  }
+
+  // Non-HVHZ Coastal check
+  if (rule.jurisdictionScope === 'Non-HVHZ Coastal' && rule.maxCoastalDistanceMiles) {
+    if (env.coastalProximityMiles > rule.maxCoastalDistanceMiles) {
+      applies = false;
